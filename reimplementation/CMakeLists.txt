cmake_minimum_required(VERSION 3.16)
project(swiv3d_reimplementation VERSION 0.1.0 LANGUAGES C CXX)

add_library(swiv3d_smacker SHARED third_party/libsmacker/smacker.c)
set_target_properties(swiv3d_smacker PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)

add_library(swiv3d_core src/assets.cpp src/audio.cpp src/cd_audio.cpp src/frontend.cpp src/mission.cpp src/mission_render.cpp src/menu_render.cpp src/alpha.cpp src/hud.cpp src/flight.cpp src/combat.cpp src/vehicles.cpp src/enemy_weapons.cpp src/mission_actors.cpp src/weapons.cpp src/special_families.cpp src/weapon_actors.cpp src/input_script.cpp src/campaign.cpp src/profile_file.cpp)

target_sources(swiv3d_core PRIVATE src/bosses.cpp)
target_sources(swiv3d_core PRIVATE src/lunar_actors.cpp)
target_sources(swiv3d_core PRIVATE src/mars_actors.cpp)
target_sources(swiv3d_core PRIVATE src/later_bosses.cpp)
target_sources(swiv3d_core PRIVATE src/game_text.cpp)
target_sources(swiv3d_core PRIVATE src/ground_effects.cpp)
target_sources(swiv3d_core PRIVATE src/cutscene.cpp)
target_sources(swiv3d_core PRIVATE src/screen_grab.cpp)
target_link_libraries(swiv3d_core PRIVATE swiv3d_smacker)
target_compile_features(swiv3d_core PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(swiv3d_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(swiv3d_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(swiv3d_assets src/main.cpp)
target_link_libraries(swiv3d_assets PRIVATE swiv3d_core)

add_executable(swiv3d_frontend src/frontend_main.cpp)
target_link_libraries(swiv3d_frontend PRIVATE swiv3d_core)

add_executable(swiv3d_player src/player_main.cpp)
target_link_libraries(swiv3d_player PRIVATE swiv3d_core)
if(WIN32)
    target_sources(swiv3d_player PRIVATE src/window_win32.cpp src/audio_win32.cpp src/device_input_win32.cpp)
    target_link_libraries(swiv3d_player PRIVATE user32 gdi32 winmm xinput9_1_0)
endif()
